⭐ 1. ميثود إضافة تقييم (AddReview)
3 شروط مجمعة في ميثود واحدة Service Layer
الشروط الـ 3 المطلوبة:
1. لازم يكون مسجل في الكورس (Enrolled).
2. ممنوع يكرر التقييم لنفس الكورس.
3. التقييم من 1 لـ 5 نجوم. 
public string AddReview(ReviewWriteDTO dto, int userId)
{
    // الشرط 1: فحص التقييم بين 1 و 5 نجوم
    if (dto.Rating < 1 || dto.Rating > 5)
        return "Rating must be between 1 and 5 stars.";

    // الشرط 2: هل اليوزر مسجل في الكورس ده أصلاً؟
    var isEnrolled = _enrollmentRepo.GetAll()
        .Any(e => e.ApplicationUserId == userId && e.CourseId == dto.CourseId);

    if (!isEnrolled)
        return "You must be enrolled in this course to leave a review.";

    // الشرط 3: هل قيّم الكورس ده قبل كده؟ (منع التكرار)
    var alreadyReviewed = _reviewRepo.GetAll()
        .Any(r => r.ApplicationUserId == userId && r.CourseId == dto.CourseId);

    if (alreadyReviewed)
        return "You have already reviewed this course.";

    // الإضافة العادية:
    var review = new Review
    {
        ApplicationUserId = userId,
        CourseId = dto.CourseId,
        Rating = dto.Rating,
        Feedback = dto.Feedback,
        CreatedAt = DateTime.UtcNow
    };

    _reviewRepo.Add(review);
    _reviewRepo.SaveChanges();
    return "Done";
}
في الـ Controller (سحب الـ userId) API / Controller 
[Authorize]
[HttpPost]
public IActionResult AddReview(ReviewWriteDTO dto)
{
    int userId = int.Parse(User.FindFirst(ClaimTypes.NameIdentifier)!.Value);

    var result = _reviewService.AddReview(dto, userId);
    if (result != "Done")
        return BadRequest(result);

    return Ok(new { message = "Review added successfully" });
}
✏️ 2. تعديل وحذف الريفيو (Ownership)
ميثود UpdateReview (فحص الملكية والتسجيل) Service Layer
الشروط: اليوزر يعدل الريفيو بتاعه هو بس، ويكون لسه مسجل في الكورس. 
public bool UpdateReview(int reviewId, int userId, ReviewUpdateDTO dto)
{
    var review = _reviewRepo.GetById(reviewId);
    if (review == null) return false;

    // الشرط 1: فحص الملكية (بتاع اليوزر ده ولا لأ؟)
    if (review.ApplicationUserId != userId)
        return false; // مش بتاعك!

    // الشرط 2: هل لسه مسجل في الكورس؟
    var isEnrolled = _enrollmentRepo.GetAll()
        .Any(e => e.ApplicationUserId == userId && e.CourseId == review.CourseId);

    if (!isEnrolled) return false;

    // تعديل البيانات:
    if (dto.Rating >= 1 && dto.Rating <= 5)
        review.Rating = dto.Rating;

    review.Feedback = dto.Feedback;

    _reviewRepo.Update(review);
    _reviewRepo.SaveChanges();
    return true;
}
ميثود DeleteReview (حذف الريفيو الخاص به) Service Layer 
public bool DeleteReview(int reviewId, int userId)
{
    var review = _reviewRepo.GetById(reviewId);
    if (review == null) return false;

    // فحص الملكية:
    if (review.ApplicationUserId != userId)
        return false;

    _reviewRepo.Delete(review);
    _reviewRepo.SaveChanges();
    return true;
}
🎓 3. ميثود التسجيل في الكورس (Enroll)
منع التسجيل المكرر Service Layer 
public bool EnrollCourse(int courseId, int userId)
{
    // الشرط: هل مسجل في نفس الكورس قبل كده؟
    var alreadyEnrolled = _enrollmentRepo.GetAll()
        .Any(e => e.ApplicationUserId == userId && e.CourseId == courseId);

    if (alreadyEnrolled)
        return false; // مسجل بالفعل

    var enrollment = new Enrollment
    {
        ApplicationUserId = userId,
        CourseId = courseId,
        EnrolledAt = DateTime.UtcNow,
        CompletedLessonsCount = 0
    };

    _enrollmentRepo.Add(enrollment);
    _enrollmentRepo.SaveChanges();
    return true;
}
📑 4. جلب الدروس مرتبة (Ordered Outlines)
ترتيب الدروس حسب OrderIndex Service Layer 
public IEnumerable<LessonReadDTO> GetCourseLessons(int courseId)
{
    return _lessonRepo.GetAll()
        .Where(l => l.CourseId == courseId)
        .OrderBy(l => l.OrderIndex) // ⚠️ الترتيب المطلوب في الكيس
        .Select(l => new LessonReadDTO
        {
            Id = l.Id,
            Title = l.Title,
            DurationMinutes = l.DurationMinutes,
            ContentSummary = l.ContentSummary,
            OrderIndex = l.OrderIndex
        }).ToList();
}
