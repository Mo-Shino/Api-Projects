// 1. تسجيل الـ DbContext والـ Repos والـ Services
builder.Services.AddDbContext<ApplicationContext>(x =>
    x.UseSqlServer(builder.Configuration.GetConnectionString("conn")));

builder.Services.AddScoped<IAccountService, AccountService>();

// 2. تسجيل الـ Identity
builder.Services.AddIdentity<ApplicationUser, IdentityRole<int>>()
    .AddEntityFrameworkStores<ApplicationContext>()
    .AddDefaultTokenProviders();

// 3. تسجيل الـ Authentication والـ JWT Bearer
builder.Services.AddAuthentication(x =>
{
    x.DefaultAuthenticateScheme = JwtBearerDefaults.AuthenticationScheme;
    x.DefaultChallengeScheme = JwtBearerDefaults.AuthenticationScheme;
})
.AddJwtBearer(options =>
{
    options.TokenValidationParameters = new TokenValidationParameters
    {
        IssuerSigningKey = new SymmetricSecurityKey(
            Encoding.UTF8.GetBytes(builder.Configuration["JWT:Key"]!)),
        ValidateIssuerSigningKey = true,
        ValidateIssuer = false,
        ValidateAudience = false
    };
});

var app = builder.Build();

// ...
app.UseHttpsRedirection();

// 4. لازم بالترتيب ده قبل MapControllers:
app.UseAuthentication();
app.UseAuthorization();

app.MapControllers();
app.Run();

AccountService.cs (Register + Login كاملة) BL / Services
💡 افتكر خطوات التوكن: Claims ⬅️ Key ⬅️ SigningCredentials ⬅️ JwtSecurityToken ⬅️ WriteToken (وكلهم قبلهم كلمة new!).
public class AccountService : IAccountService
{
    private readonly UserManager<ApplicationUser> _userManager;
    private readonly IConfiguration _configuration;

    public AccountService(UserManager<ApplicationUser> userManager, IConfiguration configuration)
    {
        _userManager = userManager;
        _configuration = configuration;
    }

    public async Task<string> Register(RegisterDTO dTO)
    {
        var reg = new ApplicationUser
        {
            UserName = dTO.UserName,
            Email = dTO.Email
        };
        var res = await _userManager.CreateAsync(reg, dTO.Password);
        if (!res.Succeeded) return "Error in register";
        return "Done";
    }

    public async Task<string> Login(LoginDTO dTO)
    {
        var u = await _userManager.FindByNameAsync(dTO.UserName);
        if (u == null) return "error";

        var corr = await _userManager.CheckPasswordAsync(u, dTO.Password);
        if (!corr) return "error";

        // 1. الـ Claims (بنحط الـ ID والاسم جوه التوكن)
        var claims = new[]
        {
            new Claim(ClaimTypes.NameIdentifier, u.Id.ToString()),
            new Claim(ClaimTypes.Name, u.UserName!)
        };

        // 2. المفتاح والختم
        var key = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(_configuration["JWT:Key"]!));
        var cred = new SigningCredentials(key, SecurityAlgorithms.HmacSha256);

        // 3. تصميم التوكن
        var tokens = new JwtSecurityToken(
            claims: claims,
            signingCredentials: cred,
            expires: DateTime.UtcNow.AddDays(1),
            issuer: _configuration["JWT:Issuer"],
            audience: _configuration["JWT:Audience"]
        );

        // 4. طباعة التوكن كنص
        return new JwtSecurityTokenHandler().WriteToken(tokens);
    }
}
🎮 4. الـ AccountController وسحب الـ UserId
AccountController.cs API / Controllers 
[Route("api/[controller]")]
[ApiController]
public class AccountController : ControllerBase
{
    private readonly IAccountService _accountService;
    public AccountController(IAccountService accountService)
    {
        _accountService = accountService;
    }

    [HttpPost("login")] // أوعى تنسى السطر ده!
    public async Task<IActionResult> Login(LoginDTO dTO)
    {
        var es = await _accountService.Login(dTO);
        if (es == "error") return BadRequest("Error");
        return Ok(new { token = es });
    }

    [HttpPost("register")] // أوعى تنسى السطر ده!
    public async Task<IActionResult> Register(RegisterDTO dTO)
    {
        var res = await _accountService.Register(dTO);
        if (res != "Done") return BadRequest(res);
        return Ok(res);
    }
}
إزاي تسحب الـ UserId من التوكن في أي Controller؟ تريكاية الامتحان
بنفس كلمة ClaimTypes.NameIdentifier اللي حطيتها في الـ Login بتسحبها في الـ Controller بـ User.FindFirst:
[Authorize]
[HttpPut("{id}/cancel")]
public IActionResult Cancel(int id)
{
    // سحب رقم اليوزر من التوكن:
    int userId = int.Parse(User.FindFirst(ClaimTypes.NameIdentifier)!.Value);

    var res = _bookingService.Cancel(id, userId);
    if (!res) return BadRequest("Cannot cancel");
    return Ok(res);
}

2. في الـ Service (Filter + Sort + Pagination) BL / Service
المعادلة الثابتة للـ Pagination: .Skip((Page - 1) * PageSize).Take(PageSize)
public IEnumerable<ProductReadDTO> GetAll(
    int Page = 1,
    int PageSize = 10,
    string? Category = null,
    DateTime? date = null,
    string? OrderBy = null)
{
    var prods = _productRepo.GetAll();

    // 1. Filter بنص (لو اليوزر بعته)
    if (!string.IsNullOrEmpty(Category))
        prods = prods.Where(x => x.Category == Category);

    // 2. Filter بتاريخ (لو اليوزر بعته)
    if (date != null)
        prods = prods.Where(x => x.OrderDate.Date == date.Value.Date);

    // 3. Sorting / ترتيب (لو طلبه)
    if (OrderBy == "Price")
        prods = prods.OrderBy(x => x.Price);
    else if (OrderBy == "Stock")
        prods = prods.OrderBy(x => x.Stock);

    // 4. Pagination + التحويل لـ ReadDTO
    return prods
        .Skip((Page - 1) * PageSize)
        .Take(PageSize)
        .Select(x => new ProductReadDTO
        {
            Id = x.Id,
            Name = x.Name,
            Price = x.Price,
            Stock = x.Stock
        }).ToList();
}
3. في الـ Controller (GetAll) API / Controller
⚠️ أوعى تنسى علامة الاستفهام ? و = null جنب الفلاتر عشان متضربش 400 لو اليوزر سابها فاضية!
[HttpGet]
public IActionResult GetAll(
    [FromQuery] int Page = 1,
    [FromQuery] int PageSize = 10,
    [FromQuery] string? Category = null,
    [FromQuery] DateTime? date = null,
    [FromQuery] string? OrderBy = null)
{

}
🛠️ 6. أشهر الإيرورات وحلها في ثانية
1. '... is a type, which is not valid in the given context' / 'Non-invocable member' / 'Method name expected'
السبب: نسيت تكتب كلمة new قبل اسم الكلاس أو الأوبجكت!
راجع الأماكن دي وحط قبلها new:
• new Claim(...)
• new SymmetricSecurityKey(...)
• new SigningCredentials(...)
• new JwtSecurityToken(...)
• new JwtSecurityTokenHandler().WriteToken(...)
• new TokenValidationParameters { ... }
• .Select(x => new ProductReadDTO { ... })
• return Ok(new { token = es });
2. Unable to resolve service for type 'I...Service' (500 Error)
السبب: نسيت تسجل الـ Service أو الـ Repo في Program.cs.
الحل: زود builder.Services.AddScoped<IService, Service>();
3. 404 Not Found في Postman على فانكشن عليها [Authorize]
السبب: الـ Identity شغالة كوكيز ومش شايفة الـ JWT.
الحل: في Program.cs جوه AddAuthentication حط:
x.DefaultAuthenticateScheme = JwtBearerDefaults.AuthenticationScheme;
x.DefaultChallengeScheme = JwtBearerDefaults.AuthenticationScheme;
4. 400 Bad Request: "The OrderBy / Category field is required"
السبب: كتبت string OrderBy من غير علامة استفهام في الـ GetAll.
الحل: حط علامة استفهام string? OrderBy = null في الـ Controller والـ Service.
5. 500 Failed to load API definition (أول ما Swagger يفتح)
السبب: نسيت تحط [HttpPost("login")] و [HttpPost("register")] أو [HttpGet] فوق الفانكشن في الـ Controller!
6. www-authenticate: Bearer (401 Unauthorized)
السبب: الفانكشن عليها [Authorize] وبتجربها في Swagger أو من غير توكن.
الحل: افتح Postman ⬅️ تبويب Authorization ⬅️ اختار Bearer Token (مش JWT Bearer) ⬅️ الصق التوكن ودوس Send.
7. Could not send request (في Postman)
السبب: المشروع مقفول في Visual Studio، أو رقم البورت في الرابط غلط.
الحل: سيب صفحة Swagger مفتوحة، وانسخ الرابط بالبورت بتاعها بالظبط (مثلاً http://localhost:5213/products).
8. IDX10720: key size must be at least '256' bits
السبب: الـ JWT:Key في appsettings.json قصير.
الحل: طول الـ Key في appsettings.json خليه سطر طويل (أكتر من 35 حرف).
9. 'Service' does not implement interface member 'IService...'
السبب: اسم الفانكشن أو الباراميترز اللي جواها في الكلاس مختلف عن الـ Interface.
الحل: انسخ سطر الفانكشن من الكلاس حطه في الـ Interface بالملّي.
10. Error in register (لما بتعمل يوزر جديد)
السبب: الباسورد مفيهوش شروط الـ Identity.
الحل: اكتب باسورد فيه (كابيتال + سمول + رقم + رمز) زي: Shino@123
